## Technical Details

### 進程內 Flow State 管理

```python
import time
from dataclasses import dataclass

# 模組層級 dict 存放進行中的 OAuth flows
_pending_oauth_flows: dict[str, OAuthFlowState] = {}
_FLOW_TIMEOUT = 300  # 5 minutes

def _cleanup_expired_flows():
    """清理超時的 OAuth flows"""
    now = time.time()
    expired = [k for k, v in _pending_oauth_flows.items() if now - v.created_at > _FLOW_TIMEOUT]
    for k in expired:
        del _pending_oauth_flows[k]
```

### 修改: POST /api/mcp/servers/{name}/auth

@app.post("/api/mcp/servers/{name}/auth")
async def auth_mcp_server(name: str, profile: Optional[str] = None):
    base_url = os.environ.get("HERMES_BASE_URL")

    if base_url:
        # === Dashboard Mode (非阻塞) ===
        _cleanup_expired_flows()

        cfg = _get_server_config(name, profile)
        if not cfg or not cfg.get("url"):
            return {"ok": False, "error": "Server not found or not HTTP transport"}

        cfg["auth"] = "oauth"

        from tools.mcp_oauth import build_dashboard_oauth_flow
        flow_state = await build_dashboard_oauth_flow(
            server_name=name,
            server_url=cfg["url"],
            base_url=base_url,
            oauth_config=cfg.get("oauth"),
        )

        _pending_oauth_flows[flow_state.state] = flow_state

        # 持久化 auth: oauth 到 config
        _save_server_config(name, cfg, profile)

        return {
            "ok": True,
            "authorization_url": flow_state.authorization_url,
            "mode": "dashboard",
        }
    else:
        # === CLI Mode (維持原本阻塞行為) ===

### 新增: GET /api/mcp/oauth/callback/{server_name}

```python
@app.get("/api/mcp/oauth/callback/{server_name}")
async def mcp_oauth_callback(server_name: str, code: str, state: str):
    _cleanup_expired_flows()

    flow = _pending_oauth_flows.get(state)
    if not flow:
        return HTMLResponse("<h2>❌ Invalid or expired OAuth state</h2>", status_code=400)

    if flow.server_name != server_name:
        return HTMLResponse("<h2>❌ Server name mismatch</h2>", status_code=400)

    try:
        from tools.mcp_oauth import exchange_code_for_token
        await exchange_code_for_token(flow, code)
        del _pending_oauth_flows[state]

        return HTMLResponse("""
        <html><body style="font-family:system-ui;text-align:center;padding:60px">
          <h2>✅ 授權成功 / Authorization Successful</h2>
          <p>此視窗將自動關閉 / This window will close automatically</p>
          <script>setTimeout(()=>window.close(), 1500)</script>
        </body></html>
        """)
    except Exception as e:
        return HTMLResponse(f"<h2>❌ Token exchange failed: {e}</h2>", status_code=500)
```

### 新增: GET /api/mcp/servers/{name}/auth/status

```python
@app.get("/api/mcp/servers/{name}/auth/status")
async def mcp_auth_status(name: str):
    from tools.mcp_oauth import HermesTokenStorage
    storage = HermesTokenStorage(name)

    has_tokens = storage.has_cached_tokens()
    if not has_tokens:
        return {"authenticated": False, "tools_count": 0, "expires_at": None}

    tokens = await storage.get_tokens()
    expires_at = None
    if hasattr(storage, '_get_expires_at'):
        expires_at = storage._get_expires_at()

    # 從 runtime MCP server state 取得 tool count
    tools_count = 0
    from tools.mcp_tool import get_mcp_server
    server = get_mcp_server(name)
    if server and server.tools:
        tools_count = len(server.tools)

    return {
        "authenticated": True,
        "tools_count": tools_count,
        "expires_at": expires_at,
    }
```
